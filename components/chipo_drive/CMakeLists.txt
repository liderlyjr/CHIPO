# CHIPO DRIVE: copilot mode for the car (voice on/off, motion detection, personality).
set(CHIPO_DRIVE_SRCS
    chipo_drive_voice.cpp
    chipo_drive_motion.cpp
    chipo_drive_personality.cpp
    chipo_drive.cpp)

# Pre-recorded voice clips (OGG/Opus, 48 kHz mono) found in clips/ are embedded in the firmware and
# looked up by name. No clips: CHIPO shows the text instead. See docs/CHIPO_DRIVE.md.
# (No CONFIGURE_DEPENDS: ESP-IDF also runs this file in script mode. After adding clips: idf.py reconfigure.)
file(GLOB CHIPO_DRIVE_CLIPS ${CMAKE_CURRENT_SOURCE_DIR}/clips/*.ogg)
set(CLIP_TABLE "${CMAKE_CURRENT_BINARY_DIR}/chipo_drive_clips_table.cpp")
set(CLIP_TABLE_TEXT "// Generated by CMake: the clips embedded from components/chipo_drive/clips.\n")
string(APPEND CLIP_TABLE_TEXT "#include \"chipo_drive_clips.h\"\n#include <cstring>\n")
set(CLIP_ENTRIES "")
foreach(CLIP ${CHIPO_DRIVE_CLIPS})
    get_filename_component(CLIP_FILE ${CLIP} NAME)
    get_filename_component(CLIP_ID ${CLIP} NAME_WE)
    string(MAKE_C_IDENTIFIER "${CLIP_FILE}" CLIP_SYM)
    string(APPEND CLIP_TABLE_TEXT "extern const unsigned char ${CLIP_SYM}_start[] asm(\"_binary_${CLIP_SYM}_start\");\n")
    string(APPEND CLIP_TABLE_TEXT "extern const unsigned char ${CLIP_SYM}_end[] asm(\"_binary_${CLIP_SYM}_end\");\n")
    string(APPEND CLIP_ENTRIES "    {\"${CLIP_ID}\", ${CLIP_SYM}_start, ${CLIP_SYM}_end},\n")
endforeach()
string(APPEND CLIP_TABLE_TEXT "namespace {\nstruct Entry { const char* id; const unsigned char* start; const unsigned char* end; };\n")
string(APPEND CLIP_TABLE_TEXT "const Entry kClips[] = {\n${CLIP_ENTRIES}    {nullptr, nullptr, nullptr}};\n}  // namespace\n")
string(APPEND CLIP_TABLE_TEXT "std::string_view ChipoDriveClip(const char* id) {\n    if (id == nullptr) return {};\n    for (const Entry& e : kClips) {\n        if (e.id && std::strcmp(e.id, id) == 0) return {reinterpret_cast<const char*>(e.start), static_cast<size_t>(e.end - e.start)};\n    }\n    return {};\n}\n")
# Written now (not at generate time) and only when it changed, so the build is not redone for nothing.
if(NOT CMAKE_BUILD_EARLY_EXPANSION)  # the early requirements scan runs in script mode: nothing to write there
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
    set(OLD_TABLE_TEXT "")
    if(EXISTS ${CLIP_TABLE})
        file(READ ${CLIP_TABLE} OLD_TABLE_TEXT)
    endif()
    if(NOT "${OLD_TABLE_TEXT}" STREQUAL "${CLIP_TABLE_TEXT}")
        file(WRITE ${CLIP_TABLE} "${CLIP_TABLE_TEXT}")
    endif()
endif()

idf_component_register(
    SRCS ${CHIPO_DRIVE_SRCS} ${CLIP_TABLE}
    INCLUDE_DIRS "include"
    EMBED_FILES ${CHIPO_DRIVE_CLIPS}
    REQUIRES chipo_face lvgl log
    PRIV_REQUIRES esp_system
)
